def run(command, log=None, env=None, timeout=180):
    command=list(map(str,command))
    started=time.time_ns()
    process=subprocess.Popen(command,cwd=ROOT,env=env or ENV,stdout=subprocess.PIPE,stderr=subprocess.PIPE,
                             text=True,creationflags=subprocess.CREATE_NO_WINDOW)
    try: stdout,stderr=process.communicate(timeout=timeout)
    except subprocess.TimeoutExpired:
        process.kill();process.wait();raise
    result=subprocess.CompletedProcess(command,process.returncode,stdout,stderr)
    result.pid=process.pid;result.started_ns=started
    if log:
        Path(log).write_text(result.stdout + '\nSTDERR:\n' + result.stderr, encoding='utf8')
    if result.returncode:
        raise RuntimeError((command, result.returncode, result.stdout[-6000:], result.stderr[-3000:]))
    return result

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def fixtures():
    result=[]
    for n in (1000,10000,100000):
        for distribution in ('uniform','clustered'):
            state=73211
            def next_int(limit):
                nonlocal state
                state=(state*1664525+1013904223)&0xffffffff
                return (state>>8)%limit
            boxes=[]
            for i in range(n):
                x=i%8*125+next_int(30) if distribution=='clustered' else next_int(1000)
                y=i%8*125+next_int(30) if distribution=='clustered' else next_int(1000)
                boxes.append((x,y,x+next_int(5)+1,y+next_int(5)+1))
            queries=[]
            for i in range(64):
                x,y=next_int(1000),next_int(1000)
                width=(0,20,200,1000)[i%4]
                queries.append((x,y,x+width,y+width))
            path=ROOT/f'fixture-{n}-{distribution}.txt'
            path.write_text(str(n)+'\n'+'\n'.join(' '.join(map(str,b)) for b in boxes+queries)+'\n',encoding='ascii')
            result.append((n,distribution,path))
    return result

def prepare():
    # Immutable snapshot created before any candidate edits. Copy only source, never user scratch.
    for stage in ('branchless','vector'):
        destination=ROOT/f'{stage}-source/src/Tedd.RTree'
        destination.mkdir(parents=True,exist_ok=False)
        for file in (ROOT/'baseline-source/src/Tedd.RTree').iterdir():
            if file.suffix in ('.cs','.csproj'): shutil.copyfile(file,destination/file.name)
    for name in ('Rectangle.cs','Rectangle2D.Generic.cs','Box.cs'):
        path=ROOT/f'branchless-source/src/Tedd.RTree/{name}'
        source=path.read_text()
        start=source.index('    public bool Intersects(')
        end=source.index('    internal ',start)
        source=source[:start]+source[start:end].replace(' &&',' &')+source[end:]
        path.write_text(source)
    path=ROOT/'vector-source/src/Tedd.RTree/RTree.cs'
    source=path.read_text()
    source='using System.Runtime.Intrinsics;\nusing System.Runtime.InteropServices;\n\n'+source
    source=source.replace('return Search(_root, bounds, results);','''if (Vector256.IsHardwareAccelerated)
            return SearchVector(_root, bounds, Vector256.Create(bounds.MaxX, bounds.MaxY, -bounds.MinX, -bounds.MinY), results);
        return Search(_root, bounds, results);''',1)
    at=source.index('    private bool Remove(Node node,')
    vector='''    private static int SearchVector(Node node, in Rectangle bounds, Vector256<double> query, List<T> results)
    {
        // Rectangle has four sequential doubles. Cast only the validated active prefix.
        ReadOnlySpan<Vector256<double>> entries = MemoryMarshal.Cast<Rectangle, Vector256<double>>(node.Boxes.AsSpan(0, node.Count));
        Vector256<double> signs = Vector256.Create(0.0, 0.0, -0.0, -0.0);
        int found = 0;
        if (node.Leaf)
        {
            for (int i = 0; i < entries.Length; i++)
            {
                if (Vector256.LessThanOrEqualAll(Vector256.Xor(entries[i], signs), query))
                {
                    results.Add(node.Values![i]);
                    found++;
                }
            }
        }
        else
        {
            for (int i = 0; i < entries.Length; i++)
            {
                if (Vector256.LessThanOrEqualAll(Vector256.Xor(entries[i], signs), query))
                    found += bounds.Contains(node.Boxes[i])
                        ? CollectValues(node.Children![i]!, results)
                        : SearchVector(node.Children![i]!, bounds, query, results);
            }
        }
        return found;
    }

'''
    path.write_text(source[:at]+vector+source[at:])
    for stage in ('baseline','branchless','vector'):
        build(stage)
    common_caller()
    for stage in ('baseline','branchless','vector'):
        stage_caller(stage)
        tests(stage)

    final=phase.startswith('final-')
    if final: phase=phase[6:]
    if phase=='exploration': names, rounds, seed=['baseline','branchless','vector'],3,181041
    elif phase=='collection': names,rounds,seed=['baseline','runs','ranges'],3,181043
    elif phase=='refinement': names,rounds,seed=['baseline','ranges-fixed','ranges-raw'],3,181045
    elif phase=='confirmation': names,rounds,seed=['baseline','selected'],5,181042
    elif phase=='generic': names,rounds,seed=['baseline','selected'],2,181046
    elif phase=='wrappers': names,rounds,seed=['baseline','selected'],2,181047
    elif phase=='tiered': names,rounds,seed=['baseline','selected'],3,181048
    elif phase=='chunks': names,rounds,seed=['baseline','selected','chunked'],3,181050
    elif phase=='fill': names,rounds,seed=['baseline','selected','filled'],3,181051
    else: raise RuntimeError('Unknown phase')
    if final:
        names=[source_stage if s=='selected' else s for s in names]
        seed+=100

    if phase in ('chunks','fill'): fixtures=[f for f in fixtures if f[2]==100000]
    jobs=[(b,*f,s) for b in range(rounds) for f in fixtures for s in names]
    random.Random(seed).shuffle(jobs)
    destination=EVIDENCE/destination_phase
    destination.mkdir(parents=True,exist_ok=False)
    warm=final and phase=='tiered'
    worker_suffix='warm-worker' if warm else 'worker'
    identities={s:{'caller':digest(ROOT/f'{s}-{worker_suffix}/Bench.dll'),

    if len({v['caller'] for v in identities.values()})!=1: raise RuntimeError('Caller drift')

    shutil.copyfile(__file__,destination/'driver.py.txt')
    samples=[]; launches=[]
    for index,(block,dim,coord,n,distribution,mode,stage) in enumerate(jobs):
        command=[DOTNET,'exec','--fx-version','10.0.13',str(ROOT/f'{stage}-{worker_suffix}/Bench.dll'),str(dim),coord,str(n),distribution,mode]
        if warm: command+=['measure-warm']

        for row in csv.reader(stdout.splitlines()):

def summarize(destination):
    samples=json.loads((destination/'samples.json').read_text())
    result=[]
    keys=sorted({(s['dimension'],s['coordinate'],s['size'],s['distribution'],s['mode'],s['stage']) for s in samples})
    for key in keys:
        rows=[s for s in samples if (s['dimension'],s['coordinate'],s['size'],s['distribution'],s['mode'],s['stage'])==key]
        for op in sorted({r['operation'] for r in rows}|{'aggregate','post-aggregate'}):
            if op in ('aggregate','post-aggregate'):
                prefix='query' if op=='aggregate' else 'mutated'
                if not any(r['operation'].startswith(prefix) for r in rows): continue
                data=[sum(r['us'] for r in rows if r['block']==b and r['operation'].startswith(prefix)) for b in sorted({r['block'] for r in rows})]
                allocations=[]
            else:
                data=[r['us'] for r in rows if r['operation']==op]
                allocations=[r['allocated'] for r in rows if r['operation']==op]
            result.append(dict(zip(('dimension','coordinate','size','distribution','mode','stage'),key),operation=op,
                median_us=statistics.median(data),min_us=min(data),max_us=max(data),
                median_allocated=statistics.median(allocations) if allocations and all(a is not None for a in allocations) else None))
    (destination/'summary.json').write_text(json.dumps(result,indent=2))
    for row in result:
        if row['operation']=='aggregate': print(row,flush=True)
